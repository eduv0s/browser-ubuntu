Here is the build plan for your Kahoot testing tool, followed by the complete HTML/CSS/JS code.


### Build Plan


**1. Core Architecture & Philosophy**

- **Single HTML File:** All HTML, CSS, and JavaScript contained within one `index.html` for easy deployment and use.

- **No Backend Required:** The tool runs entirely in the browser. It simulates bot behavior and API interactions. *Note: True auto-answering for all public Kahoot games is impossible without a backend proxy due to CORS and API restrictions. Therefore, the tool uses a sophisticated simulation mode for demonstration & testing, which you can explain to students as a proof-of-concept.*

- **Two Primary Modes:** Toggled via an editorial, asymmetric tab system at the top.

    - **Mode 1: The Agent (Auto-Answerer):** Demonstrates reading the question and selecting the correct answer (simulated).

    - **Mode 2: The Spammer (Bot Flooder):** Creates and manages multiple bot participants in a lobby.


**2. Visual Design System (Breaking the AI Mold)**

- **Typography:**

    - Headings: `Cormorant Garamond` (serif, editorial, sharp).

    - Body/Action: `Inter` (clean, technical, but not generic).

- **Color Palette:**

    - Background: `#F9F8F6` (warm off-white, paper-like).

    - Primary Text: `#1A1A1A` (almost black).

    - Accent/Highlight: `#E85D04` (vibrant, urgent orange).

    - Secondary Action: `#27272A` (deep gray).

- **Layout Constraints:**

    - **No Cards:** Information is separated by vertical rhythm (margin/padding) and font weights.

    - **No Borders:** No visible `border` properties on containers or headings.

    - **Asymmetry:** The control panels use an 35/65 or 40/60 split instead of 50/50. Lists and logs are right-aligned or extend to the edge.


**3. Functional Logic (Simulated & Educational)**

- **Kahoot Code & Username:** Input fields for lobby PIN and bot username(s).

- **Agent Mode (Auto-Answer):**

    - *Simulation:* When "Join & Auto-Answer" is clicked, it simulates joining. Then, a "Question" appears (mock data). The "AI" (using a mock LLM call or a local answer key) selects the correct option visually.

    - *API Key Ready:* An input field for OpenRouter/DeepSeek key is provided. In a real deployment with a backend, you'd connect it here. For the frontend demo, it calls a simulated AI function.

- **Spammer Mode (Bot Flooder):**

    - Generate bots: Input number of bots (1-100). Click "Flood Lobby".

    - Each bot gets a unique name (Bot_1, Bot_2, etc.) and a randomized "connection" status.

    - **Real-time Dashboard:** A live list of active bots with "Disconnect" buttons. A score monitor and join speed slider.

    - The bot names appear in a real-time log.

- **Educational Disclaimer:** Clear disclaimer that this is for authorized testing and anti-cheat research only.


**4. Component Breakdown**

- **Header Section:** Title, description, and toggle for Agent/Spammer.

- **Agent Panel:** API Key input, Kahoot PIN, Username, "Deploy Agent" button, simulated question area, answer log.

- **Spammer Panel:** PIN input, bot count slider, join speed slider, "Start Flood" button, active bots dashboard (flex/grid list), global disconnect button.

- **Footer:** Teacher/Testing disclaimer.


**5. Code Implementation Strategy**

- Use modern CSS Grid (magazine style) for layout.

- Vanilla JavaScript for DOM manipulation and state management.

- No external dependencies (except Google Fonts).

- Simulated network requests with `setTimeout` to mimic real bot behavior without actually spamming Kahoot's servers (responsible practice).

  function addLog(elementId, msg) {

    const logDiv = document.getElementById(elementId);

    if(logDiv) {

      const p = document.createElement('div');

      p.innerHTML = `> ${new Date().toLocaleTimeString()} · ${msg}`;

  // ---- SPAMMER LOGIC (simulated bot creation, monitor) ----

  function renderBotDashboard() {

    const container = document.getElementById('botsContainer');

    const counterSpan = document.getElementById('botCounter');

    if(!container) return;

    if(activeBots.length === 0) {

      container.innerHTML = '<div style="padding: 1rem 0; color: #71717A;">no bots active — start flood</div>';

      if(counterSpan) counterSpan.innerText = '0 connected';

      return;

    }

    counterSpan.innerText = `${activeBots.length} connected`;

    let html = '';

    activeBots.forEach(bot => {

      html += `

        <div class="bot-item" data-id="${bot.id}">

          <span class="bot-name">🤖 ${escapeHtml(bot.name)}</span>

          <span class="bot-score">${bot.score} pts</span>

          <button class="disconnect-bot" data-id="${bot.id}">kick</button>

        </div>

      `;

    });

    container.innerHTML = html;

    // attach disconnect events

    document.querySelectorAll('.disconnect-bot').forEach(btn => {

      btn.addEventListener('click', (e) => {

        const id = parseInt(btn.getAttribute('data-id'));

        disconnectBotById(id);

      });

    });

  }


  function disconnectBotById(id) {

    const index = activeBots.findIndex(b => b.id === id);

    if(index !== -1) {

      const removed = activeBots[index];

      activeBots.splice(index,1);

      renderBotDashboard();

      addLog('spamLog', `Bot "${removed.name}" disconnected from lobby.`);

    }

  }


  function disconnectAllBots() {

    if(floodInterval) { clearInterval(floodInterval); floodInterval = null; }

    const count = activeBots.length;

    activeBots = [];

    renderBotDashboard();

    addLog('spamLog', `Global kill: ${count} bot(s) removed.`);

  }


  function addBot(name) {

    const newId = Date.now() + Math.floor(Math.random() * 10000);

    const newBot = {

      id: newId,

      name: name,

      score: Math.floor(Math.random() * 2800),

      status: 'active'

    };

    activeBots.push(newBot);

    renderBotDashboard();

    addLog('spamLog', `Bot "${name}" joined lobby (PIN ${document.getElementById('spamPin').value || '****'})`);

    // simulate random answer pattern for "dashboard score live"

    setInterval(() => {

      const botStillHere = activeBots.find(b => b.id === newId);

      if(botStillHere && Math.random() > 0.65) {

        botStillHere.score += Math.floor(Math.random() * 450);

        renderBotDashboard();

      }

    }, 4000);

  }


  function floodLobby() {

    const pin = document.getElementById('spamPin').value.trim();

    if(!pin) { addLog('spamLog','❌ please enter a game PIN (testing mode)'); return; }

    let botAmount = parseInt(document.getElementById('botCount').value);

    if(isNaN(botAmount) || botAmount < 1) botAmount = 1;

    if(botAmount > 120) botAmount = 120;

    const speed = parseInt(document.getElementById('joinSpeed').value);

    let index = 0;

    if(floodInterval) clearInterval(floodInterval);

    addLog('spamLog', `🚀 Starting flood of ${botAmount} bots to PIN ${pin} (simulated join speed ${speed}ms)`);

    floodInterval = setInterval(() => {

      if(index >= botAmount) {

        clearInterval(floodInterval);

        floodInterval = null;

        addLog('spamLog', `✅ Flood finished. ${botAmount} bots joined.`);

        return;

      }

      const botName = `Flood_${Math.floor(Math.random()*9000)+1000}_${index}`;

      addBot(botName);

      index++;

    }, speed);

  }

      // simulate AI decision after 1 sec
